def main():
    # create the Bridges object, set credentials
    bridges = Bridges(YOUR_ASSSIGNMENT_NUMBER, "YOUR_USER_ID", "YOUR_API_KEY")

    l1.font_size = fsize
    l1.text = "Charlotte"
    sc.add_symbol(l1);

    c2 = Circle(locx=2.3514, locy=48.8576, r=1.)
    c2.stroke_color = scol
    c2.fillcolor = fcol
    c2.font_size = fsize
    c2.strok_size = str;
    sc.add_symbol(c2);
